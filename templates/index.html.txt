<script>
    // Get OS Icon
    function getOsIcon(osName) {
        const lowerOS = osName.toLowerCase();
        if (lowerOS.includes('win')) return '<i class="fa-brands fa-windows os-icon text-info"></i>';
        if (lowerOS.includes('kali') || lowerOS.includes('linux')) return '<i class="fa-brands fa-linux os-icon text-warning"></i>';
        if (lowerOS.includes('mac')) return '<i class="fa-brands fa-apple os-icon text-light"></i>';
        return '<i class="fa-solid fa-desktop os-icon text-secondary"></i>';
    }

    // Determine Progress Bar Color
    function getProgressBarColor(value) {
        if (value < 50) return 'bg-success';
        if (value < 80) return 'bg-warning';
        return 'bg-danger';
    }

    // Extract percentage value (e.g., "15%" -> 15)
    function parsePercentage(valStr) {
        const match = valStr.match(/(\d+(\.\d+)?)/);
        return match ? parseFloat(match[0]) : 0;
    }

    async function fetchData() {
        try {
            const res = await fetch('/api/clients');
            const data = await res.json();
            const tbody = document.getElementById('clientTableBody');
            
            // Keep default loading message if no data
            if (Object.keys(data).length === 0) return;
            
            tbody.innerHTML = ''; 
            const now = Math.floor(Date.now() / 1000);

            for (const [id, info] of Object.entries(data)) {
                // Timeout Logic: Offline if no heartbeat in 30 seconds
                const isOnline = (now - info.last_seen) <= 30; 
                
                const statusHtml = isOnline 
                    ? `<span class="status-dot online-dot"></span> <span class="text-success fw-bold">Online</span>` 
                    : `<span class="status-dot offline-dot"></span> <span class="text-danger fw-bold">Offline</span>`;
                
                // Format time using standard en-GB for 24h format
                const timeStr = new Date(info.last_seen * 1000).toLocaleTimeString('en-GB');
                const osHtml = `${getOsIcon(info.os)} ${info.os}`;
                
                // Process CPU & RAM data
                const cpuVal = parsePercentage(info.cpu);
                const ramVal = parsePercentage(info.ram);
                const cpuColor = getProgressBarColor(cpuVal);
                const ramColor = getProgressBarColor(ramVal);

                tbody.innerHTML += `
                    <tr>
                        <td class="ps-4 fw-bold code-font">${id}</td>
                        <td class="code-font text-muted">${info.ip}</td>
                        <td>${osHtml}</td>
                        <td>
                            <div class="d-flex justify-content-between align-items-center mb-1">
                                <span class="code-font" style="font-size: 0.8rem;">${info.cpu}</span>
                            </div>
                            <div class="progress">
                                <div class="progress-bar ${cpuColor}" role="progressbar" style="width: ${cpuVal}%"></div>
                            </div>
                        </td>
                        <td>
                            <div class="d-flex justify-content-between align-items-center mb-1">
                                <span class="code-font" style="font-size: 0.8rem;">${info.ram}</span>
                            </div>
                            <div class="progress">
                                <div class="progress-bar ${ramColor}" role="progressbar" style="width: ${ramVal}%"></div>
                            </div>
                        </td>
                        <td>${statusHtml}</td>
                        <td class="pe-4 code-font text-muted">${timeStr}</td>
                    </tr>
                `;
            }
        } catch (error) {
            console.error("API Connection Error:", error);
        }
    }
    
    // Auto-refresh every 2 seconds
    setInterval(fetchData, 2000);
    fetchData();
</script>
